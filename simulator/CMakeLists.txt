cmake_minimum_required(VERSION 3.25)
project(simulator VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(simulatorBuildServer "Build the browser application server" ON)
option(simulatorSanitizers "Enable address and undefined behavior checks" OFF)

find_package(Threads REQUIRED)
find_package(nlohmann_json 3.12 CONFIG REQUIRED)
if(simulatorBuildServer)
  find_package(Boost 1.90 CONFIG REQUIRED)
endif()

add_library(simulatorOptions INTERFACE)
if(MSVC)
  target_compile_options(simulatorOptions INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(simulatorOptions INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(simulatorSanitizers AND NOT MSVC)
  target_compile_options(simulatorOptions INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(simulatorOptions INTERFACE -fsanitize=address,undefined)
endif()

add_library(simulatorCore src/blockRegistry.cpp src/world.cpp src/simulator.cpp src/project.cpp src/pistons.cpp src/devices.cpp src/containers.cpp src/hoppers.cpp)
target_include_directories(simulatorCore PUBLIC include)
target_link_libraries(simulatorCore PUBLIC nlohmann_json::nlohmann_json simulatorOptions Threads::Threads)
target_compile_definitions(simulatorCore PUBLIC SIMULATOR_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")

add_executable(simulatorCli apps/cli/main.cpp)
target_link_libraries(simulatorCli PRIVATE simulatorCore)
if(simulatorBuildServer)
  add_executable(simulatorServer apps/server/main.cpp apps/server/server.cpp)
  target_link_libraries(simulatorServer PRIVATE simulatorCore Boost::headers)
  target_compile_definitions(simulatorServer PRIVATE BOOST_ERROR_CODE_HEADER_ONLY SIMULATOR_WEB_DIR="${CMAKE_CURRENT_SOURCE_DIR}/apps/web/dist")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(simulatorTests tests/coreTests.cpp)
  target_link_libraries(simulatorTests PRIVATE simulatorCore)
  add_test(NAME simulatorCoreTests COMMAND simulatorTests)
endif()
