cmake_minimum_required(VERSION 3.25)
project(verimcPhysicalAdapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(simulatorBuildServer OFF CACHE BOOL "No web server required" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build adapter only" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../simulator" simulator)
add_executable(physicalAdapter main.cpp)
target_link_libraries(physicalAdapter PRIVATE simulatorCore)
target_include_directories(physicalAdapter PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/simulator/generated")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" adapterSourceSha256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
target_compile_definitions(physicalAdapter PRIVATE VERIMC_ADAPTER_SHA256="${adapterSourceSha256}")
