cmake_minimum_required(VERSION 3.25)
project(verimc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(verimcSanitizers "Enable ASan and UBSan" OFF)
option(verimcBuildCompiler "Build the source compiler" ON)
option(verimcBuildJson "Build the .vmcl JSON adapter" ON)
find_package(Boost 1.90 CONFIG REQUIRED)

add_library(verimcIr src/diagnostic.cpp src/logicTypes.cpp src/logicGraph.cpp)
target_include_directories(verimcIr PUBLIC include)
target_link_libraries(verimcIr PUBLIC Boost::headers)
set(verimcTargets verimcIr)
if(verimcBuildJson OR verimcBuildCompiler)
  find_package(nlohmann_json 3.12 CONFIG REQUIRED)
endif()
if(verimcBuildJson)
  add_library(verimcJson src/vmclJson.cpp)
  target_link_libraries(verimcJson PUBLIC verimcIr PRIVATE nlohmann_json::nlohmann_json)
  list(APPEND verimcTargets verimcJson)
endif()
if(verimcBuildCompiler)
  find_package(Java 17 REQUIRED COMPONENTS Runtime)
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  include(cmake/antlr.cmake)
  add_library(verimcCore src/ast.cpp src/compiler.cpp)
  target_link_libraries(verimcCore PUBLIC verimcIr PRIVATE nlohmann_json::nlohmann_json verimcParser OpenSSL::Crypto)
  list(APPEND verimcTargets verimcCore)
endif()
foreach(verimcTarget IN LISTS verimcTargets)
  if(MSVC)
    target_compile_options(${verimcTarget} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${verimcTarget} PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  endif()
endforeach()
if(verimcSanitizers AND NOT MSVC)
  if(verimcBuildCompiler)
    list(APPEND verimcTargets verimcParser antlr4_static)
  endif()
  foreach(verimcTarget IN LISTS verimcTargets)
    target_compile_options(${verimcTarget} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${verimcTarget} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
if(verimcBuildCompiler AND verimcBuildJson)
  add_executable(verimcCli apps/cli/main.cpp)
  target_link_libraries(verimcCli PRIVATE verimcCore verimcJson nlohmann_json::nlohmann_json)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(verimcIrTests tests/logicIrTests.cpp)
  target_link_libraries(verimcIrTests PRIVATE verimcIr)
  add_test(NAME logicIr COMMAND verimcIrTests)
  if(verimcBuildJson)
    add_executable(verimcJsonTests tests/vmclJsonTests.cpp)
    target_link_libraries(verimcJsonTests PRIVATE verimcJson)
    add_test(NAME vmclJson COMMAND verimcJsonTests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/legacyCounter.vmcl)
  endif()
  if(verimcBuildCompiler)
    add_executable(verimcCompilerApiTests tests/compilerApiTests.cpp)
    target_link_libraries(verimcCompilerApiTests PRIVATE verimcCore)
    add_test(NAME compilerApi COMMAND verimcCompilerApiTests ${CMAKE_CURRENT_SOURCE_DIR}/examples/valid/counter.vmc)
  endif()
  if(TARGET verimcCli)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME compilerRegression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compilerTests.py $<TARGET_FILE:verimcCli>)
  endif()
endif()
