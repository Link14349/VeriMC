cmake_minimum_required(VERSION 3.25)
project(verimc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(verimcSanitizers "Enable ASan and UBSan" OFF)
find_package(Java 17 REQUIRED COMPONENTS Runtime)
find_package(Boost 1.90 CONFIG REQUIRED)
find_package(nlohmann_json 3.12 CONFIG REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
include(cmake/antlr.cmake)
add_library(verimcCore src/ast.cpp src/compiler.cpp src/logicGraph.cpp)
target_include_directories(verimcCore PUBLIC include)
target_link_libraries(verimcCore PUBLIC nlohmann_json::nlohmann_json Boost::headers PRIVATE verimcParser OpenSSL::Crypto)
if(MSVC)
  target_compile_options(verimcCore PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(verimcCore PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
endif()
if(verimcSanitizers AND NOT MSVC)
  foreach(verimcTarget verimcCore verimcParser antlr4_static)
    target_compile_options(${verimcTarget} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${verimcTarget} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
add_executable(verimcCli apps/cli/main.cpp)
target_link_libraries(verimcCli PRIVATE verimcCore)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME compilerRegression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compilerTests.py $<TARGET_FILE:verimcCli>)
endif()
